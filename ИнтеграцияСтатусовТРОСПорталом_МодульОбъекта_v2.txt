Функция СоздатьHTTPСоединение()

	Адрес = СокрЛП(АдресПортала);
	Если ПустаяСтрока(Адрес) Тогда
		ВызватьИсключение "Не указан адрес портала.";
	КонецЕсли;

	HTTPS = Ложь;
	Если Лев(НРег(Адрес), 8) = "https://" Тогда
		HTTPS = Истина;
		ХостИПорт = Сред(Адрес, 9);
		ПортПоУмолчанию = 443;
	ИначеЕсли Лев(НРег(Адрес), 7) = "http://" Тогда
		ХостИПорт = Сред(Адрес, 8);
		ПортПоУмолчанию = 80;
	Иначе
		ВызватьИсключение
			"Адрес портала должен начинаться с http:// или https://.";
	КонецЕсли;

	Если Прав(ХостИПорт, 1) = "/" Тогда
		ХостИПорт = Лев(ХостИПорт, СтрДлина(ХостИПорт) - 1);
	КонецЕсли;

	ПозицияДвоеточия = Найти(ХостИПорт, ":");
	Если ПозицияДвоеточия > 0 Тогда
		Хост = Лев(ХостИПорт, ПозицияДвоеточия - 1);
		СтрокаПорта = Сред(ХостИПорт, ПозицияДвоеточия + 1);
		Попытка
			Порт = Число(СтрокаПорта);
		Исключение
			ВызватьИсключение "Некорректный порт в адресе портала.";
		КонецПопытки;
	Иначе
		Хост = ХостИПорт;
		Порт = ПортПоУмолчанию;
	КонецЕсли;

	Если HTTPS Тогда
		SSL = Новый ЗащищенноеСоединениеOpenSSL();
		Возврат Новый HTTPСоединение(Хост, Порт, , , , 60, SSL);
	Иначе
		Возврат Новый HTTPСоединение(Хост, Порт, , , , 60);
	КонецЕсли;

КонецФункции

Функция ВыполнитьHTTPЗапрос(
	Метод,
	Путь,
	ТелоJSON = "",
	ИспользоватьАвторизацию = Истина)

	Соединение = СоздатьHTTPСоединение();
	ЗапросHTTP = Новый HTTPЗапрос(Путь);
	ЗапросHTTP.Заголовки.Вставить("Accept", "application/json");

	Если ИспользоватьАвторизацию Тогда
		ЗапросHTTP.Заголовки.Вставить(
			"Authorization",
			"Bearer " + ТокенДоступа);
	КонецЕсли;

	Если Метод = "POST" Тогда
		ЗапросHTTP.Заголовки.Вставить("Content-Type", "application/json");
		ЗапросHTTP.УстановитьТелоИзСтроки(ТелоJSON, КодировкаТекста.UTF8);
		HTTPОтвет = Соединение.ОтправитьДляОбработки(ЗапросHTTP);
	ИначеЕсли Метод = "GET" Тогда
		HTTPОтвет = Соединение.Получить(ЗапросHTTP);
	Иначе
		ВызватьИсключение "Неподдерживаемый HTTP-метод: " + Метод;
	КонецЕсли;

	ТекстОтвета = HTTPОтвет.ПолучитьТелоКакСтроку();
	JSONОтвет = Неопределено;
	Если НЕ ПустаяСтрока(ТекстОтвета) Тогда
		Попытка
			JSONОтвет = JSONВЗначение(ТекстОтвета);
		Исключение
			JSONОтвет = Неопределено;
		КонецПопытки;
	КонецЕсли;

	Возврат Новый Структура(
		"Код,Текст,JSON",
		HTTPОтвет.КодСостояния,
		ТекстОтвета,
		JSONОтвет);

КонецФункции

Процедура Авторизоваться() Экспорт

	ПроверитьНастройкиИнтеграции();

	Тело = Новый Структура;
	Тело.Вставить("login", СокрЛП(ЛогинИнтеграции));
	Тело.Вставить("password", ПарольИнтеграции);

	Ответ = ВыполнитьHTTPЗапрос(
		"POST",
		"/api/1c/auth/sign-in",
		ЗначениеВJSON(Тело),
		Ложь);

	Если Ответ.Код <> 200 Тогда
		ВызватьИсключение
			"Ошибка авторизации в Portal. HTTP "
			+ Строка(Ответ.Код)
			+ ". "
			+ Ответ.Текст;
	КонецЕсли;

	Если Ответ.JSON = Неопределено Тогда
		ВызватьИсключение "Portal вернул некорректный JSON авторизации.";
	КонецЕсли;

	ДанныеАвторизации = ПолучитьJSONЗначение(
		Ответ.JSON,
		"data",
		Неопределено);

	Если ДанныеАвторизации = Неопределено
		ИЛИ ДанныеАвторизации = NULL Тогда

		ВызватьИсключение
			"В ответе Portal отсутствует data авторизации.";

	КонецЕсли;

	ТокенДоступа = ПолучитьJSONСтроку(
		ДанныеАвторизации,
		"accessToken");
	СрокМС = ПолучитьJSONЗначение(
		ДанныеАвторизации,
		"accessTokenExpiration",
		0);

	Если ПустаяСтрока(ТокенДоступа) Тогда
		ВызватьИсключение "В ответе Portal отсутствует accessToken.";
	КонецЕсли;

	Если СрокМС <= 60000 Тогда
		ВызватьИсключение "Portal вернул некорректный срок действия accessToken.";
	КонецЕсли;

	ТокенДействителенДо = ТекущаяДата() + (СрокМС / 1000) - 60;

КонецПроцедуры

Функция ВыполнитьAPIЗапрос(
	Метод,
	Путь,
	ТелоJSON = "",
	ПовторятьПосле401 = Истина)

	Если ПустаяСтрока(ТокенДоступа)
		ИЛИ ТекущаяДата() >= ТокенДействителенДо Тогда
		Авторизоваться();
	КонецЕсли;

	Ответ = ВыполнитьHTTPЗапрос(Метод, Путь, ТелоJSON, Истина);
	Если Ответ.Код = 401 И ПовторятьПосле401 Тогда
		СброситьАвторизацию();
		Авторизоваться();
		Возврат ВыполнитьAPIЗапрос(Метод, Путь, ТелоJSON, Ложь);
	КонецЕсли;

	Возврат Ответ;

Функция ПолучитьОписаниеОшибкиAPI(Ответ)

	Описание = "HTTP " + Строка(Ответ.Код);
	Если Ответ.JSON = Неопределено Тогда
		Возврат Описание;
	КонецЕсли;

	ОшибкаJSON = ПолучитьJSONЗначение(Ответ.JSON, "error", Неопределено);
	Если ОшибкаJSON = Неопределено Тогда
		Возврат Описание;
	КонецЕсли;

	КодОшибки = ПолучитьJSONСтроку(ОшибкаJSON, "code");
	ТекстОшибки = ПолучитьJSONСтроку(ОшибкаJSON, "message");
	Если НЕ ПустаяСтрока(КодОшибки) Тогда
		Описание = Описание + " " + КодОшибки;
	КонецЕсли;
	Если НЕ ПустаяСтрока(ТекстОшибки) Тогда
		Описание = Описание + ": " + ТекстОшибки;
	КонецЕсли;

	Возврат Описание;

Функция ДатаВISO(ДатаЗначение)

	Если НЕ ЗначениеЗаполнено(ДатаЗначение) Тогда
		Возврат "";
	КонецЕсли;

	Смещение = ТекущаяДата() - ТекущаяУниверсальнаяДата();
	ДатаUTC = ДатаЗначение - Смещение;

	Возврат Формат(ДатаUTC, "ДФ=yyyy-MM-ddTHH:mm:ss") + "Z";

		Если СтадияРодителя = Перечисления.СтадииДокументовТРО.Новый Тогда

		Возврат "IN_PROGRESS";

		Если НЕ ДокументВыполнения.Подтвержден Тогда
			ВсеПодтверждены = Ложь;

		КонецЕсли;

	Если СтадияРодителя = Перечисления.СтадииДокументовТРО.Завершен

		|	Заявка.Комментарий КАК ИдентификаторПортала
		|ИЗ
		|	Документ.ЗаявкаНаВозвратТРО КАК Заявка
		|ГДЕ
		|	Заявка.Ссылка = &СсылкаДокумента";

	Иначе

		ВызватьИсключение
			"Неподдерживаемый movement_type: "
			+ ВидДвижения;

	КонецЕсли;


	Запрос.УстановитьПараметр(
		"СсылкаДокумента",
		СсылкаДокумента);

	Выборка =
		Запрос.Выполнить().Выбрать();


	//////////////////////////////////////////////////////////
	// ДОКУМЕНТ НЕ НАЙДЕН
	//////////////////////////////////////////////////////////

	Если НЕ Выборка.Следующий() Тогда

		Возврат Новый Структура(
			"Найден,СсылкаДокумента,Стадия1С,СтадияPortal",
			Ложь,
			СсылкаДокумента,
			Неопределено,
			"");

	КонецЕсли;


	//////////////////////////////////////////////////////////
	// CROSS-CHECK PORTAL IDENTITY
	//////////////////////////////////////////////////////////

	Если ВидДвижения = "INSTALL" Тогда

		НомерЭлектронногоДокумента =
			СокрЛП(
				Строка(
					Выборка.ИдентификаторПортала));

		Если НЕ ПустаяСтрока(
			НомерЭлектронногоДокумента)
			И НомерЭлектронногоДокумента
				<> СокрЛП(НомерПортала) Тогда

			ВызватьИсключение
				"GUID заказа найден, но "
				+ "НомерЭлектронногоДокумента 1С "
				+ "не соответствует номеру Portal.";

		КонецЕсли;


	ИначеЕсли ВидДвижения = "RETURN" Тогда

		Комментарий =
			Строка(
				Выборка.ИдентификаторПортала);

		Маркер =
			"PORTAL-TRO:"
			+ СокрЛП(НомерПортала)
			+ ";";

		Если Лев(
			Комментарий,
			СтрДлина(Маркер)) <> Маркер Тогда

			ВызватьИсключение
				"GUID заявки возврата найден, "
				+ "но служебный маркер Portal "
				+ "в комментарии не соответствует "
				+ "номеру Portal.";

		КонецЕсли;

	КонецЕсли;


	//////////////////////////////////////////////////////////
	// STAGE
	//////////////////////////////////////////////////////////

	ФактическиеДокументы = ПолучитьФактическиеДокументы1С(
		ВидДвижения,
		Выборка.Ссылка);
	СтадияPortal = ПолучитьИтоговуюСтадиюПортала(
		Выборка.СтадияДокументаТРО,
		ФактическиеДокументы,
		Выборка.ПометкаУдаления);


	Возврат Новый Структура(
		"Найден,СсылкаДокумента,Стадия1С,СтадияPortal,ФактическиеДокументы",
		Истина,
		Выборка.Ссылка,
		Выборка.СтадияДокументаТРО,
		СтадияPortal,
		ФактическиеДокументы);

Функция ПолучитьСтраницуОчереди(Cursor)

	Путь = "/api/1c/tro-requests/status-pending?limit="
		+ Формат(РазмерСтраницы, "ЧГ=0");

	Если НЕ ПустаяСтрока(Cursor) Тогда
		Путь = Путь
			+ "&cursor="
			+ КодироватьСтроку(
				Cursor,
				СпособКодированияСтроки.КодировкаURL);
	КонецЕсли;

	Ответ = ВыполнитьAPIЗапрос("GET", Путь);
	Если Ответ.Код <> 200 Тогда
		ВызватьИсключение
			"Не удалось получить очередь status-pending: "
			+ ПолучитьОписаниеОшибкиAPI(Ответ);
	КонецЕсли;

	Если Ответ.JSON = Неопределено Тогда
		ВызватьИсключение
			"Portal вернул пустой или некорректный JSON status-pending.";
	КонецЕсли;

	Если НЕ ПолучитьJSONЗначение(Ответ.JSON, "ok", Ложь) Тогда
		ВызватьИсключение "Portal вернул ok=false для status-pending.";
	КонецЕсли;

	Данные = ПолучитьJSONЗначение(Ответ.JSON, "data", Неопределено);
	Пагинация = ПолучитьJSONЗначение(
		Ответ.JSON,
		"pagination",
		Неопределено);

	Если Данные = Неопределено ИЛИ Пагинация = Неопределено Тогда
		ВызватьИсключение
			"В ответе status-pending отсутствуют data или pagination.";
	КонецЕсли;

	ЕстьЕще = ПолучитьJSONЗначение(Пагинация, "has_more", Ложь);
	СледующийCursor = ПолучитьJSONСтроку(Пагинация, "next_cursor");

	Если ЕстьЕще И ПустаяСтрока(СледующийCursor) Тогда
		ВызватьИсключение
			"Portal вернул has_more=true без next_cursor.";
	КонецЕсли;

	Возврат Новый Структура(
		"Данные,ЕстьЕще,СледующийCursor",
		Данные,
		ЕстьЕще,
		СледующийCursor);

Функция ОтправитьСтадиюНаПортал(
	PortalDocumentID,
	GUIDДокумента,
	СтадияPortal,
	МоментНаблюдения,
	ФактическиеДокументы)

	Тело = Новый Структура;
	Тело.Вставить("document_1c_guid", GUIDДокумента);
	Тело.Вставить("source_system", SourceSystem);
	Тело.Вставить("stage", СтадияPortal);
	Тело.Вставить("changed_at", ДатаВISO(МоментНаблюдения));
	Тело.Вставить("execution_documents", СформироватьExecutionDocumentsJSON(ФактическиеДокументы));

	Путь = "/api/1c/tro-requests/"
		+ Формат(PortalDocumentID, "ЧГ=0")
		+ "/stage";

	Ответ = ВыполнитьAPIЗапрос("POST", Путь, ЗначениеВJSON(Тело));
	Если Ответ.Код <> 200 Тогда
		ВызватьИсключение
			"Не удалось обновить one_c_stage: "
			+ ПолучитьОписаниеОшибкиAPI(Ответ);
	КонецЕсли;

	Если Ответ.JSON = Неопределено
		ИЛИ НЕ ПолучитьJSONЗначение(Ответ.JSON, "ok", Ложь) Тогда
		ВызватьИсключение
			"Portal вернул некорректный успешный ответ /stage.";
	КонецЕсли;

	ДанныеОтвета = ПолучитьJSONЗначение(
		Ответ.JSON,
		"data",
		Неопределено);

	Если ДанныеОтвета = Неопределено Тогда
		ВызватьИсключение "В ответе /stage отсутствует data.";
	КонецЕсли;

	Возврат Новый Структура(
		"Идемпотентно,СтадияИзменена,ДокументыИзменены",
		ПолучитьJSONЗначение(ДанныеОтвета, "idempotent", Ложь),
		ПолучитьJSONЗначение(ДанныеОтвета, "stage_changed", Ложь),
		ПолучитьJSONЗначение(ДанныеОтвета, "execution_documents_changed", Ложь));

Процедура ОбработатьСтрокуОчереди(
	ДанныеДокумента,
	Результат,
	ПросмотренныеДокументы)

	PortalDocumentID = ПолучитьJSONЗначение(
		ДанныеДокумента,
		"portal_document_id",
		0);
	НомерПортала = ПолучитьJSONСтроку(
		ДанныеДокумента,
		"portal_document_number");
	ВидДвижения = ВРег(СокрЛП(ПолучитьJSONСтроку(
		ДанныеДокумента,
		"movement_type")));
	GUIDДокумента = СокрЛП(ПолучитьJSONСтроку(
		ДанныеДокумента,
		"document_1c_guid"));
	Источник = ВРег(СокрЛП(ПолучитьJSONСтроку(
		ДанныеДокумента,
		"source_system")));
	СтадияПорталаДо = ВРег(СокрЛП(ПолучитьJSONСтроку(
		ДанныеДокумента,
		"one_c_stage")));

	Если PortalDocumentID <= 0 Тогда
		ВызватьИсключение "Отсутствует корректный portal_document_id.";
	КонецЕсли;
	Если ПустаяСтрока(НомерПортала) Тогда
		ВызватьИсключение "Отсутствует portal_document_number.";
	КонецЕсли;
	Если ПустаяСтрока(GUIDДокумента) Тогда
		ВызватьИсключение "Отсутствует document_1c_guid.";
	КонецЕсли;
	Если Источник <> "UP" Тогда
		ВызватьИсключение
			"Неподдерживаемый source_system: " + Источник;
	КонецЕсли;
	Если НЕ ДопустимаяСтадияПортала(СтадияПорталаДо) Тогда
		ВызватьИсключение
			"Portal вернул неизвестный one_c_stage: " + СтадияПорталаДо;
	КонецЕсли;

	КлючДокумента = Формат(PortalDocumentID, "ЧГ=0");
	Если ПросмотренныеДокументы.Получить(КлючДокумента) <> Неопределено Тогда
		ВызватьИсключение
			"Повтор portal_document_id в cursor traversal: " + КлючДокумента;
	КонецЕсли;
	ПросмотренныеДокументы.Вставить(КлючДокумента, Истина);

	Состояние1С = ПолучитьСостояниеДокумента1С(
		ВидДвижения,
		GUIDДокумента,
		НомерПортала);

	Если НЕ Состояние1С.Найден Тогда
		ВызватьИсключение "Документ 1С по GUID не найден.";
	КонецЕсли;

	МоментНаблюдения = ТекущаяДата();
	РезультатОтправки = ОтправитьСтадиюНаПортал(
		PortalDocumentID,
		GUIDДокумента,
		Состояние1С.СтадияPortal,
		МоментНаблюдения,
		Состояние1С.ФактическиеДокументы);

	Если РезультатОтправки.Идемпотентно Тогда
		Результат.Идемпотентно = Результат.Идемпотентно + 1;
		Результат.БезИзменений = Результат.БезИзменений + 1;
	ИначеЕсли РезультатОтправки.СтадияИзменена
		ИЛИ РезультатОтправки.ДокументыИзменены Тогда
		Результат.Изменено = Результат.Изменено + 1;
	Иначе
		Результат.БезИзменений = Результат.БезИзменений + 1;
	КонецЕсли;

	ТекстУспеха = НомерПортала + ": ";
	Если ПустаяСтрока(СтадияПорталаДо) Тогда
		ТекстУспеха = ТекстУспеха + "NULL";
	Иначе
		ТекстУспеха = ТекстУспеха + СтадияПорталаДо;
	КонецЕсли;
	Если РезультатОтправки.ДокументыИзменены
		И НЕ РезультатОтправки.СтадияИзменена Тогда
		ТекстУспеха = НомерПортала
			+ ": обновлены документы выполнения УП, стадия осталась "
			+ Состояние1С.СтадияPortal;
	Иначе
		ТекстУспеха = ТекстУспеха + " -> " + Состояние1С.СтадияPortal;
	КонецЕсли;

	ДобавитьСтрокуОписания(Результат, ТекстУспеха);
	ЗаписатьСобытиеЖурнала(
		УровеньЖурналаРегистрации.Информация,
		ТекстУспеха);

Функция СформироватьИтог(Результат)

	Возврат
		"Синхронизация стадий завершена."
		+ Символы.ПС + "Получено: " + Строка(Результат.Получено)
		+ Символы.ПС + "Проверено: " + Строка(Результат.Проверено)
		+ Символы.ПС + "Изменено: " + Строка(Результат.Изменено)
		+ Символы.ПС + "Без изменений: " + Строка(Результат.БезИзменений)
		+ Символы.ПС + "Идемпотентно: " + Строка(Результат.Идемпотентно)
		+ Символы.ПС + "Ошибок: " + Строка(Результат.Ошибок);

Функция ВыполнитьСинхронизациюСтатусов() Экспорт

	ИнициализироватьТехническиеПараметры();
	ПроверитьНастройкиИнтеграции();

	Результат = Новый Структура(
		"Получено,Проверено,Изменено,БезИзменений,Идемпотентно,Ошибок,Описание",
		0,
		0,
		0,
		0,
		0,
		0,
		"");

	Cursor = "";
	НомерСтраницы = 0;
	ПросмотренныеCursor = Новый Соответствие;
	ПросмотренныеДокументы = Новый Соответствие;

	Пока Истина Цикл

		НомерСтраницы = НомерСтраницы + 1;
		Если НомерСтраницы > МаксимумСтраниц Тогда
			ВызватьИсключение
				"Превышен защитный лимит cursor pagination: "
				+ Строка(МаксимумСтраниц) + " страниц.";
		КонецЕсли;

		Если НЕ ПустаяСтрока(Cursor) Тогда
			Если ПросмотренныеCursor.Получить(Cursor) <> Неопределено Тогда
				ВызватьИсключение
					"Portal повторил уже использованный cursor.";
			КонецЕсли;
			ПросмотренныеCursor.Вставить(Cursor, Истина);
		КонецЕсли;

		Страница = ПолучитьСтраницуОчереди(Cursor);

		Для Каждого ДанныеДокумента Из Страница.Данные Цикл

			Результат.Получено = Результат.Получено + 1;
			Результат.Проверено = Результат.Проверено + 1;

			Попытка
				ОбработатьСтрокуОчереди(
					ДанныеДокумента,
					Результат,
					ПросмотренныеДокументы);
			Исключение
				Результат.Ошибок = Результат.Ошибок + 1;

				НомерПортала = ПолучитьJSONСтроку(
					ДанныеДокумента,
					"portal_document_number");
				PortalDocumentID = ПолучитьJSONСтроку(
					ДанныеДокумента,
					"portal_document_id");
				ВидДвижения = ПолучитьJSONСтроку(
					ДанныеДокумента,
					"movement_type");
				GUIDДокумента = ПолучитьJSONСтроку(
					ДанныеДокумента,
					"document_1c_guid");

				ТекстОшибки =
					НомерПортала
					+ " [id=" + PortalDocumentID
					+ ", movement_type=" + ВидДвижения
					+ ", guid=" + GUIDДокумента
					+ "]: " + ОписаниеОшибки();

				ДобавитьСтрокуОписания(Результат, "Ошибка: " + ТекстОшибки);
				ЗаписатьСобытиеЖурнала(
					УровеньЖурналаРегистрации.Ошибка,
					ТекстОшибки);
			КонецПопытки;

		КонецЦикла;

		Если НЕ Страница.ЕстьЕще Тогда
			Прервать;
		КонецЕсли;

		Если ПустаяСтрока(Страница.СледующийCursor) Тогда
			ВызватьИсключение
				"Portal вернул has_more=true без next_cursor.";
		КонецЕсли;
		Если Страница.СледующийCursor = Cursor Тогда
			ВызватьИсключение "Portal повторил текущий cursor.";
		КонецЕсли;

		Cursor = Страница.СледующийCursor;

	КонецЦикла;

	Итог = СформироватьИтог(Результат);
	Если ПустаяСтрока(Результат.Описание) Тогда
		Результат.Описание = Итог;
	Иначе
		Результат.Описание = Итог
			+ Символы.ПС + Символы.ПС
			+ Результат.Описание;
	КонецЕсли;

	Возврат Результат;
